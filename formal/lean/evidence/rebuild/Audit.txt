theorem CentralDynamics.exact_expectation_hasSum : ∀ (T : ℝ),
  HasSum (fun n => ↑T ^ n * CentralDynamics.scalarCoefficient n) (CentralDynamics.expectation T) :=
fun T =>
  have h := ExponentialPilot.time_series_hasSum (↑T) CentralDynamics.Gplus CentralDynamics.Gminus CentralDynamics.DC;
  have he :=
    Eq.mpr
      (id
        (congrFun'
          (congrFun'
            (congrArg (@HasSum ℂ ℕ Complex.instAddCommMonoid PseudoMetricSpace.toUniformSpace.toTopologicalSpace)
              (funext fun n => congrArg (HMul.hMul (↑T ^ n)) (CentralDynamics.scalarCoefficient.eq_1 n)))
            ((NormedSpace.exp (↑T • CentralDynamics.Gplus) * CentralDynamics.DC *
                NormedSpace.exp (↑T • CentralDynamics.Gminus))
              ConcreteH.initialState ConcreteH.initialState))
          (SummationFilter.unconditional ℕ)))
      (HasSum.map h CentralDynamics.evaluation CentralDynamics.evaluation_continuous);
  have hv :=
    Eq.mpr
      (id
        (congrArg (fun _a => _a = CentralDynamics.expectation T)
          (CentralDynamics.evaluation_apply
            (NormedSpace.exp (↑T • CentralDynamics.Gplus) * CentralDynamics.DC *
              NormedSpace.exp (↑T • CentralDynamics.Gminus)))))
      (Eq.mpr
        (id
          (congrArg
            (fun _a =>
              (NormedSpace.exp (↑T • _a) * CentralDynamics.DC * NormedSpace.exp (↑T • CentralDynamics.Gminus))
                  ConcreteH.initialState ConcreteH.initialState =
                CentralDynamics.expectation T)
            CentralDynamics.generator_plus))
        (Eq.mpr
          (id
            (congrArg
              (fun _a =>
                (NormedSpace.exp (↑T • Complex.I • ConcreteH.complexH) * CentralDynamics.DC * NormedSpace.exp (↑T • _a))
                    ConcreteH.initialState ConcreteH.initialState =
                  CentralDynamics.expectation T)
              CentralDynamics.generator_minus))
          (of_eq_true
            (Eq.trans
              (congrFun'
                (congrArg Eq
                  (congrFun'
                    (congrFun'
                      (congr
                        (congrArg HMul.hMul
                          (congrFun'
                            (congrArg HMul.hMul
                              (NormedSpace.exp.congr_simp (↑T • Complex.I • ConcreteH.complexH)
                                ((↑T * Complex.I) • ConcreteH.complexH) (smul_smul (↑T) Complex.I ConcreteH.complexH)))
                            CentralDynamics.DC))
                        (NormedSpace.exp.congr_simp (↑T • -Complex.I • ConcreteH.complexH)
                          (-(↑T * Complex.I) • ConcreteH.complexH)
                          (Eq.trans (smul_smul (↑T) (-Complex.I) ConcreteH.complexH)
                            (congrFun' (congrArg HSMul.hSMul (mul_neg (↑T) Complex.I)) ConcreteH.complexH))))
                      ConcreteH.initialState)
                    ConcreteH.initialState))

                    NormedSpace.exp (-(↑T * Complex.I) • ConcreteH.complexH))
                  ConcreteH.initialState ConcreteH.initialState))
              (eq_self
                ((NormedSpace.exp ((↑T * Complex.I) • ConcreteH.complexH) * CentralDynamics.DC *
                    NormedSpace.exp (-(↑T * Complex.I) • ConcreteH.complexH))
                  ConcreteH.initialState ConcreteH.initialState))))));

theorem CentralDynamics.exact_series_split : ∀ (T : ℝ),
  CentralDynamics.expectation T = ↑T ^ 4 + ∑' (n : ℕ), ↑T ^ (n + 5) * CentralDynamics.scalarCoefficient (n + 5) :=
fun T =>
  have h := CentralDynamics.exact_expectation_hasSum T;
  have hs := Summable.sum_add_tsum_nat_add 5 (HasSum.summable h);
  Eq.symm
    (Eq.mp
      (congrArg
        (fun _a =>
          _a + ∑' (i : ℕ), ↑T ^ (i + 5) * CentralDynamics.scalarCoefficient (i + 5) = CentralDynamics.expectation T)
        (CentralDynamics.polynomial_through_four T))
      (Eq.mp
        (congrArg
          (fun _a =>
            ∑ i ∈ Finset.range 5, ↑T ^ i * CentralDynamics.scalarCoefficient i +
                ∑' (i : ℕ), ↑T ^ (i + 5) * CentralDynamics.scalarCoefficient (i + 5) =
              _a)
          (HasSum.tsum_eq h))
        hs))
